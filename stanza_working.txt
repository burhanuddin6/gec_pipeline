import contextlib
import stanza

# --- 1. Initialize Pipeline ---
print("Initializing Stanza pipeline...")
with contextlib.redirect_stdout(open(os.devnull, 'w')):

print("✅ Pipelines initialized successfully.")
print("-" * 60)

# --- 2. Define the Input Sentence ---
# A sentence with a variety of word types
urdu_sentence = "میں یونیورسٹی میں ایک نیا کورس پڑھ رہا ہوں۔"
# Translation: "I am studying a new course in the university."
print(f"Input Sentence: {urdu_sentence}\n")
print("-" * 60)

# --- 3. Process the Sentence to create the objects ---
stanza_doc = nlp_stanza(urdu_sentence)
print("✅ Objects created by processing the sentence.")
print("-" * 60)
